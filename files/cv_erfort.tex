% The left column carries the date / qualifier in a muted small typeface; the
% right column hosts a regular paragraph block that may contain \\ line
% breaks, \par paragraphs, or short lists.
\newcommand{\cvitem}[2]{%
  \par\nopagebreak[2]%
  \noindent
  \begin{minipage}[t]{\cvleftw}\raggedright\color{cvmuted}\small\vspace{0pt}\strut #1\end{minipage}%
  \hspace{\cvgap}%
  \begin{minipage}[t]{\cvrightw}\raggedright\vspace{0pt}\strut #2\end{minipage}%
  \par\vspace{10pt plus 3pt minus 2pt}%
}
